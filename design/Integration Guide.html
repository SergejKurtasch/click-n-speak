Конвертируй SVG → PNG

rumps принимает только PNG. Используем встроенный macOS инструмент.
💡 На macOS есть встроенный rsvg-convert или qlmanage. Проще всего использовать Homebrew + librsvg или Python с cairosvg.
Вариант A — Скрипт Python (уже есть venv в проекте)
scripts/convert_icons.py 
#!/usr/bin/env python3
"""Конвертирует все SVG иконки в PNG для rumps.
Запуск: python scripts/convert_icons.py
"""
import subprocess, sys
from pathlib import Path

# Установи cairosvg если нет:
# pip install cairosvg
try:
    import cairosvg
except ImportError:
    subprocess.run([sys.executable, "-m", "pip", "install", "cairosvg"])
    import cairosvg

ROOT = Path(__file__).parent.parent / "assets" / "icons"

# Menubar иконки — 22×22 px (retina = 44×44 логически @2x)
for svg in (ROOT / "menubar").glob("*.svg"):
    out = svg.with_suffix(".png")
    cairosvg.svg2png(url=str(svg), write_to=str(out), output_width=44, output_height=44)
    print(f"✓ menubar/{svg.name} → {out.name}")

# Menu иконки — 16×16 px
for svg in (ROOT / "menu").glob("*.svg"):
    out = svg.with_suffix(".png")
    cairosvg.svg2png(url=str(svg), write_to=str(out), output_width=32, output_height=32)
    print(f"✓ menu/{svg.name} → {out.name}")

print("Готово!")

Обнови utils.py

Добавь функции для получения путей к новым иконкам
src/utils.py — добавить после get_menu_icon_path() 
def get_menubar_icon_path(state: str = "idle") -> Path:
    """Возвращает путь к PNG иконке для menu bar.
    state: 'idle' | 'recording' | 'processing'
    """
    app_bundle = _get_app_bundle()
    if app_bundle:
        p = Path(app_bundle) / "Contents" / "Resources" / "icons" / "menubar" / f"{state}.png"
        if p.exists():
            return p
    # Dev mode fallback
    p = ROOT / "assets" / "icons" / "menubar" / f"{state}.png"
    if p.exists():
        return p
    # Fallback к старой иконке
    return get_menu_icon_path()


def get_menu_item_icon_path(name: str) -> Path | None:
    """Возвращает путь к PNG иконке пункта меню (или None если не найдена).
    Пример: get_menu_item_icon_path('languages') → assets/icons/menu/languages.png
    """
    app_bundle = _get_app_bundle()
    if app_bundle:
        p = Path(app_bundle) / "Contents" / "Resources" / "icons" / "menu" / f"{name}.png"
        if p.exists():
            return p
    p = ROOT / "assets" / "icons" / "menu" / f"{name}.png"
    return p if p.exists() else None

def set_menubar_state(self, state: str) -> None:
    """Меняет иконку в menu bar. state: 'idle'|'recording'|'processing'"""
    icon = get_menubar_icon_path(state)
    self.icon = str(icon)

def _icon(name: str) -> dict:

Добавь иконки в сборку (.app bundle)

Чтобы иконки попали в .app при py2app
В setup.py или build script — добавь в data_files
# setup.py
DATA_FILES = [
    ...,
    ("icons/menubar", glob("assets/icons/menubar/*.png")),
    ("icons/menu",    glob("assets/icons/menu/*.png")),
]
✅ После этого get_menubar_icon_path() и get_menu_item_icon_path() автоматически найдут иконки как в dev-режиме, так и в собранном .app.
